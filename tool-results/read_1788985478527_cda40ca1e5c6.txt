   146→/**
   147→ * Genera los próximos `cantidad` días (hoy incluido) según la fecha de
   148→ * Costa Rica. Solo debe ejecutarse EN CLIENTE (tras el mount) para no
   149→ * romper la hidratación: depende del reloj del dispositivo.
   150→ */
   151→function generarDias(cantidad: number): DiaOption[] {
   152→  const hoyCR = new Date().toLocaleDateString("en-CA", { timeZone: "America/Costa_Rica" });
   153→  const partes = hoyCR.split("-");
   154→  const base = new Date(Date.UTC(Number(partes[0]), Number(partes[1]) - 1, Number(partes[2])));
   155→  const fmtSemana = new Intl.DateTimeFormat("es-CR", { weekday: "short", timeZone: "UTC" });
   156→  const fmtDia = new Intl.DateTimeFormat("es-CR", { day: "numeric", timeZone: "UTC" });
   157→  const dias: DiaOption[] = [];
   158→  for (let i = 0; i < cantidad; i++) {
   159→    const fecha = new Date(base.getTime() + i * 86_400_000);
   160→    dias.push({
   161→      iso: fecha.toISOString().slice(0, 10),
   162→      weekday: fmtSemana.format(fecha).replace(".", ""),
   163→      diaMes: fmtDia.format(fecha),
   164→      esDomingo: fecha.getUTCDay() === 0,
   165→    });
   166→  }
   167→  return dias;
   168→}

   170→/** «Dra. Mariana Solís» → «MS» · «Sin preferencia» → «SP». */
   171→function inicialesProfesional(label: string): string {
   172→  const limpio = label.replace(/^Dra?\.?\s+/i, "");
   173→  const iniciales = limpio
   174→    .split(/\s+/)
   175→    .filter(Boolean)
   176→    .slice(0, 2)
   177→    .map((palabra) => palabra.charAt(0).toUpperCase());
   178→  return iniciales.join("") || "LV";
   179→}

   240→export default function BookingSection() {
   241→  const [paso, setPaso] = useState(1);
   242→  const [form, setForm] = useState<Formulario>(FORM_INICIAL);
   243→  const [errores, setErrores] = useState<Record<string, string>>({});
   244→  const [dias, setDias] = useState<DiaOption[]>([]);
   245→  const [disponibilidad, setDisponibilidad] = useState<{ closed: boolean; taken: string[] } | null>(null);
   246→  const [cargandoDispo, setCargandoDispo] = useState(false);
   247→  const [errorDispo, setErrorDispo] = useState(false);
   248→  const [enviando, setEnviando] = useState(false);
   249→  const [confirmacion, setConfirmacion] = useState<Confirmacion | null>(null);
   250→  const [servicioDestacado, setServicioDestacado] = useState<string | null>(null);
   251→  const tarjetaRef = useRef<HTMLDivElement>(null);
   252→
   253→  /* Actualiza un campo y limpia su error asociado. */
   254→  const actualizar = useCallback(<K extends keyof Formulario>(campo: K, valor: Formulario[K]) => {
   255→    setForm((prev) => ({ ...prev, [campo]: valor }));
   256→    setErrores((prev) => {
   257→      if (!(campo in prev)) return prev;
   258→      const siguientes = { ...prev };
   259→      delete siguientes[campo as string];
   260→      return siguientes;
   261→    });
   262→  }, []);
   263→
   264→  /* Días del calendario: se generan tras el mount (hidratación segura). */
   265→  useEffect(() => {
   266→    setDias(generarDias(DIAS_HORIZONTE));
   267→  }, []);
   268→
   269→  /* Mantiene el asistente a la vista al cambiar de paso o al confirmar. */
   270→  useEffect(() => {
   271→    tarjetaRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" });
   272→  }, [paso, confirmacion]);
   273→
   274→  /* Preselección de servicio desde las tarjetas de #servicios:
   275→     el enlace «Agendar» dispara un CustomEvent global; aquí lo recibimos,
   276→     volvemos al paso 1, precargamos el servicio y avisamos con un toast.
   277→     Los setters de React son estables, así que el listener se registra una
   278→     sola vez (el scroll al ancla lo hace el navegador de forma nativa). */
   279→  useEffect(() => {
   280→    const preseleccionar = (evento: Event) => {
   281→      const detail = (evento as CustomEvent<{ serviceId?: unknown }>).detail;
   282→      const serviceId = typeof detail?.serviceId === "string" ? detail.serviceId : "";
   283→      const servicio = services.find((s) => s.id === serviceId);
   284→      if (!servicio) return; // id desconocido: se ignora el evento
   285→
   286→      setConfirmacion(null); // si había una confirmación en pantalla, volver al asistente
   287→      setPaso(1);
   288→      setErrores({});
   289→      setForm((prev) => ({ ...prev, service: servicio.id }));
   290→      setServicioDestacado(servicio.id);
   291→      toast.success(`Servicio seleccionado: ${servicio.label} — continúa con tu reserva`, {
   292→        description: "Revisa los datos de tu mascota y elige el día de tu cita.",
   293→      });
   294→    };
   295→
   296→    window.addEventListener(EVENTO_PRESELECCION_SERVICIO, preseleccionar);
   297→    return () =>
   298→      window.removeEventListener(EVENTO_PRESELECCION_SERVICIO, preseleccionar);
   299→  }, []);
   300→
   301→  /* El resaltado del servicio preseleccionado dura ~2 s. */
   302→  useEffect(() => {
   303→    if (!servicioDestacado) return;
   304→    const temporizador = setTimeout(() => setServicioDestacado(null), DURACION_DESTELLO_MS);
   305→    return () => clearTimeout(temporizador);
   306→  }, [servicioDestacado]);
   307→
   308→  const consultarDisponibilidad = useCallback(async (fecha: string) => {
   309→    setCargandoDispo(true);
   310→    setErrorDispo(false);
   311→    try {
   312→      const res = await fetch(`/api/availability?date=${encodeURIComponent(fecha)}`, { cache: "no-store" });
   313→      const datos = (await res.json().catch(() => null)) as {
   314→        ok?: boolean;
   315→        closed?: boolean;
   316→        taken?: string[];
   317→        error?: string;
   318→      } | null;
   319→      if (!res.ok || !datos?.ok) throw new Error(datos?.error ?? "Error de disponibilidad");
   320→      setDisponibilidad({ closed: Boolean(datos.closed), taken: Array.isArray(datos.taken) ? datos.taken : [] });
   321→    } catch {
   322→      setDisponibilidad(null);
   323→      setErrorDispo(true);
   324→    } finally {
   325→      setCargandoDispo(false);
   326→    }
   327→  }, []);
   328→
   329→  const elegirDia = (iso: string) => {
   330→    setForm((prev) => ({ ...prev, date: iso, timeSlot: "" }));
   331→    setErrores((prev) => {
   332→      if (!prev.date && !prev.timeSlot) return prev;
   333→      const siguientes = { ...prev };
   334→      delete siguientes.date;
   335→      delete siguientes.timeSlot;
   336→      return siguientes;
   337→    });
   338→    void consultarDisponibilidad(iso);
   339→  };
   340→
   341→  /* Validación manual por paso (campos requeridos visibles + error inline). */
   342→  const validarPaso = (numero: number): Record<string, string> => {
   343→    const encontrados: Record<string, string> = {};
   344→    if (numero === 1) {
   345→      if (form.petName.trim().length < 2) {
   346→        encontrados.petName = "Escribe el nombre de tu mascota (mínimo 2 letras).";
   347→      }
   348→      if (!form.species) encontrados.species = "Selecciona la especie de tu mascota.";
   349→      if (!form.service) encontrados.service = "Selecciona el servicio que tu mascota necesita.";
   350→      if (form.breed.trim().length > 60) encontrados.breed = "La raza no puede superar los 60 caracteres.";
   351→    }
   352→    if (numero === 3) {
   353→      if (!form.date) encontrados.date = "Elige el día de tu cita.";
   354→      if (!form.timeSlot) encontrados.timeSlot = "Elige un horario disponible.";
   355→    }
   356→    if (numero === 4) {
   357→      if (form.tutorName.trim().length < 3) encontrados.tutorName = "Escribe tu nombre completo.";
   358→      if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) {
   359→        encontrados.email = "Escribe un correo electrónico válido.";
   360→      }
   361→      const tel = form.phone.trim();
   362→      if (tel.length < 8 || !/^[\d\s+()-]+$/.test(tel)) {
   363→        encontrados.phone = "Escribe un teléfono válido (mínimo 8 dígitos).";
   364→      }
   365→      if (form.notes.length > 500) encontrados.notes = "Las notas no pueden superar los 500 caracteres.";
   366→      if (!form.consent) encontrados.consent = "Necesitamos tu autorización para coordinar la cita.";
   367→    }
   368→    return encontrados;
   369→  };
   370→
   371→  const avanzar = () => {
   372→    const encontrados = validarPaso(paso);
   373→    if (Object.keys(encontrados).length > 0) {
   374→      setErrores(encontrados);
   375→      return;
   376→    }
   377→    setErrores({});
   378→    setPaso((prev) => Math.min(PASOS.length, prev + 1));
   379→  };
   380→
   381→  const retroceder = () => {
   382→    setErrores({});
   383→    setPaso((prev) => Math.max(1, prev - 1));
   384→  };
   385→
   386→  /* Enter en pasos 1–3 solo avanza (nunca envía); en el paso 4 envía. */
   387→  const enviarReserva = async (evento: FormEvent<HTMLFormElement>) => {
   388→    evento.preventDefault();
   389→    if (paso < PASOS.length) {
   390→      avanzar();
   391→      return;
   392→    }
   393→
   394→    const encontrados = validarPaso(4);
   395→    if (Object.keys(encontrados).length > 0) {
   396→      setErrores(encontrados);
   397→      return;
   398→    }
   399→    setErrores({});
   400→    setEnviando(true);
   401→
   402→    try {
   403→      const payload: AppointmentInput = {
   404→        petName: form.petName.trim(),
   405→        species: form.species as Especie,
   406→        breed: form.breed.trim(),
   407→        ageYears: form.ageYears,
   408→        service: form.service,
   409→        preferredVet: form.preferredVet,
   410→        date: form.date,
   411→        timeSlot: form.timeSlot,
   412→        tutorName: form.tutorName.trim(),
   413→        email: form.email.trim(),
   414→        phone: form.phone.trim(),
   415→        notes: form.notes.trim(),
   416→        consent: true,
   417→      };
   418→
   419→      const res = await fetch("/api/appointments", {
   420→        method: "POST",
   421→        headers: { "Content-Type": "application/json" },
   422→        body: JSON.stringify(payload),
   423→      });
   424→      const json = (await res.json().catch(() => null)) as {
   425→        ok?: boolean;
   426→        code?: string;
   427→        error?: string;
   428→        fieldErrors?: Record<string, string>;
   429→        appointment?: { date: string; timeSlot: string; service: string; petName: string; tutorName: string };
   430→      } | null;
   431→
   432→      if (res.status === 409) {
   433→        // La franja se ocupó mientras agendábamos: volver al paso 3 y refrescar.
   434→        toast.error(json?.error ?? "Lo sentimos, esa franja acaba de ser reservada. Elige otro horario.");
   435→        setForm((prev) => ({ ...prev, timeSlot: "" }));
   436→        setErrores((prev) => {
   437→          const siguientes = { ...prev };
   438→          delete siguientes.timeSlot;
   439→          return siguientes;
   440→        });
   441→        setPaso(3);
   442→        if (form.date) void consultarDisponibilidad(form.date);
   443→        return;
   444→      }
   445→
   446→      if (!res.ok || !json?.ok || !json.code || !json.appointment) {
   447→        if (json?.fieldErrors) setErrores(json.fieldErrors);
   448→        toast.error(json?.error ?? "No pudimos registrar tu reserva. Revisa los datos e intenta de nuevo.");
   449→        return;
   450→      }
   451→
   452→      setConfirmacion({ code: json.code, ...json.appointment });
   453→      toast.success("¡Reserva confirmada! Te esperamos.");
   454→    } catch {
   455→      toast.error("Tuvimos un problema de conexión. Verifica tu internet e intenta de nuevo.");
   456→    } finally {
   457→      setEnviando(false);
   458→    }
   459→  };
   460→
   461→  const reiniciar = () => {
   462→    setForm(FORM_INICIAL);
   463→    setErrores({});
   464→    setPaso(1);
   465→    setDisponibilidad(null);
   466→    setErrorDispo(false);
   467→    setCargandoDispo(false);
   468→    setConfirmacion(null);
   469→  };
   470→
   471→  const servicioElegido = services.find((servicio) => servicio.id === form.service);
   472→  const profesionalElegido = vets.find((vet) => vet.id === form.preferredVet);
   473→  const tituloPasoActual = PASOS[paso - 1]?.titulo ?? "";
   474→  const pasoActual = confirmacion ? PASOS.length : paso;
   475→
   476→  /* Enlace de WhatsApp con confirmación prellenada (solo tras confirmar). */
   477→  const waConfirmHref = confirmacion
   478→    ? `https://wa.me/${site.whatsappNumber}?text=${encodeURIComponent(

   480→          confirmacion.date,
   481→        )} a las ${confirmacion.timeSlot}.`,
   482→      )}`
   483→    : "#";
   484→
   485→  return (
   486→    <MotionConfig reducedMotion="user">
   487→      {/* Sonner: montado aquí porque el layout solo monta el Toaster de radix;
   488→          así los toasts de sonner (preselección, 409, éxito) sí se ven. */}
   489→      <ToasterSonner position="top-center" closeButton />
   490→      <section id="agendar" aria-labelledby="titulo-agendar" className="bg-background py-20 lg:py-24">
   491→        <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">
   492→          {/* Encabezado de la sección */}
   493→          <div className="mx-auto mb-12 max-w-2xl text-center lg:mb-16">
   494→            <p className="text-sm font-bold uppercase tracking-[0.2em] text-brand-teal">
   495→              Reserva en menos de 1 minuto
   496→            </p>
   497→            <h2
   498→              id="titulo-agendar"
   499→              className="mt-3 text-3xl font-extrabold tracking-tight text-foreground sm:text-4xl"
   500→            >
   501→              Agenda la cita de tu compañero
   502→            </h2>
   503→            <p className="mt-4 text-lg text-muted-foreground">
   504→              Cuatro pasos sencillos para reservar la consulta de tu mascota senior: sin registros ni llamadas
   505→              obligatorias.
   506→            </p>
   507→          </div>
   508→
   509→          <div className="grid gap-8 lg:grid-cols-5">
   510→            {/* ── Panel informativo (sticky) ── */}
   511→            <div className="lg:col-span-2">
   512→              <div className="space-y-4 lg:sticky lg:top-24">
   513→                <div className="rounded-3xl bg-primary p-8 text-primary-foreground dark:bg-brand-navy">
   514→                  <h3 className="text-xl font-bold tracking-tight">Qué pasa después de reservar</h3>
   515→                  <ul className="mt-6 space-y-5">

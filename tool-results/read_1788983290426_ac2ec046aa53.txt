   143→/**
   144→ * Genera los próximos `cantidad` días (hoy incluido) según la fecha de
   145→ * Costa Rica. Solo debe ejecutarse EN CLIENTE (tras el mount) para no
   146→ * romper la hidratación: depende del reloj del dispositivo.
   147→ */
   148→function generarDias(cantidad: number): DiaOption[] {
   149→  const hoyCR = new Date().toLocaleDateString("en-CA", { timeZone: "America/Costa_Rica" });
   150→  const partes = hoyCR.split("-");
   151→  const base = new Date(Date.UTC(Number(partes[0]), Number(partes[1]) - 1, Number(partes[2])));
   152→  const fmtSemana = new Intl.DateTimeFormat("es-CR", { weekday: "short", timeZone: "UTC" });
   153→  const fmtDia = new Intl.DateTimeFormat("es-CR", { day: "numeric", timeZone: "UTC" });
   154→  const dias: DiaOption[] = [];
   155→  for (let i = 0; i < cantidad; i++) {
   156→    const fecha = new Date(base.getTime() + i * 86_400_000);
   157→    dias.push({
   158→      iso: fecha.toISOString().slice(0, 10),
   159→      weekday: fmtSemana.format(fecha).replace(".", ""),
   160→      diaMes: fmtDia.format(fecha),
   161→      esDomingo: fecha.getUTCDay() === 0,
   162→    });
   163→  }
   164→  return dias;
   165→}

   167→/** «Dra. Mariana Solís» → «MS» · «Sin preferencia» → «SP». */
   168→function inicialesProfesional(label: string): string {
   169→  const limpio = label.replace(/^Dra?\.?\s+/i, "");
   170→  const iniciales = limpio
   171→    .split(/\s+/)
   172→    .filter(Boolean)
   173→    .slice(0, 2)
   174→    .map((palabra) => palabra.charAt(0).toUpperCase());
   175→  return iniciales.join("") || "LV";
   176→}

   240→  const [errores, setErrores] = useState<Record<string, string>>({});

   243→  const [cargandoDispo, setCargandoDispo] = useState(false);
   244→  const [errorDispo, setErrorDispo] = useState(false);
   245→  const [enviando, setEnviando] = useState(false);
   246→  const [confirmacion, setConfirmacion] = useState<Confirmacion | null>(null);
   247→  const tarjetaRef = useRef<HTMLDivElement>(null);
   248→
   249→  /* Actualiza un campo y limpia su error asociado. */
   250→  const actualizar = useCallback(<K extends keyof Formulario>(campo: K, valor: Formulario[K]) => {
   251→    setForm((prev) => ({ ...prev, [campo]: valor }));
   252→    setErrores((prev) => {
   253→      if (!(campo in prev)) return prev;
   254→      const siguientes = { ...prev };
   255→      delete siguientes[campo as string];
   256→      return siguientes;
   257→    });
   258→  }, []);
   259→
   260→  /* Días del calendario: se generan tras el mount (hidratación segura). */
   261→  useEffect(() => {
   262→    setDias(generarDias(DIAS_HORIZONTE));
   263→  }, []);
   264→
   265→  /* Mantiene el asistente a la vista al cambiar de paso o al confirmar. */
   266→  useEffect(() => {
   267→    tarjetaRef.current?.scrollIntoView({ behavior: "smooth", block: "nearest" });
   268→  }, [paso, confirmacion]);
   269→
   270→  const consultarDisponibilidad = useCallback(async (fecha: string) => {
   271→    setCargandoDispo(true);
   272→    setErrorDispo(false);
   273→    try {
   274→      const res = await fetch(`/api/availability?date=${encodeURIComponent(fecha)}`, { cache: "no-store" });
   275→      const datos = (await res.json().catch(() => null)) as {
   276→        ok?: boolean;
   277→        closed?: boolean;
   278→        taken?: string[];
   279→        error?: string;
   280→      } | null;
   281→      if (!res.ok || !datos?.ok) throw new Error(datos?.error ?? "Error de disponibilidad");
   282→      setDisponibilidad({ closed: Boolean(datos.closed), taken: Array.isArray(datos.taken) ? datos.taken : [] });
   283→    } catch {
   284→      setDisponibilidad(null);
   285→      setErrorDispo(true);
   286→    } finally {
   287→      setCargandoDispo(false);
   288→    }
   289→  }, []);
   290→
   291→  const elegirDia = (iso: string) => {
   292→    setForm((prev) => ({ ...prev, date: iso, timeSlot: "" }));
   293→    setErrores((prev) => {
   294→      if (!prev.date && !prev.timeSlot) return prev;
   295→      const siguientes = { ...prev };
   296→      delete siguientes.date;
   297→      delete siguientes.timeSlot;
   298→      return siguientes;
   299→    });
   300→    void consultarDisponibilidad(iso);
   301→  };
   302→
   303→  /* Validación manual por paso (campos requeridos visibles + error inline). */
   304→  const validarPaso = (numero: number): Record<string, string> => {
   305→    const encontrados: Record<string, string> = {};
   306→    if (numero === 1) {
   307→      if (form.petName.trim().length < 2) {
   308→        encontrados.petName = "Escribe el nombre de tu mascota (mínimo 2 letras).";
   309→      }
   310→      if (!form.species) encontrados.species = "Selecciona la especie de tu mascota.";
   311→      if (!form.service) encontrados.service = "Selecciona el servicio que tu mascota necesita.";
   312→      if (form.breed.trim().length > 60) encontrados.breed = "La raza no puede superar los 60 caracteres.";
   313→    }
   314→    if (numero === 3) {
   315→      if (!form.date) encontrados.date = "Elige el día de tu cita.";
   316→      if (!form.timeSlot) encontrados.timeSlot = "Elige un horario disponible.";
   317→    }
   318→    if (numero === 4) {
   319→      if (form.tutorName.trim().length < 3) encontrados.tutorName = "Escribe tu nombre completo.";
   320→      if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(form.email.trim())) {
   321→        encontrados.email = "Escribe un correo electrónico válido.";
   322→      }
   323→      const tel = form.phone.trim();
   324→      if (tel.length < 8 || !/^[\d\s+()-]+$/.test(tel)) {
   325→        encontrados.phone = "Escribe un teléfono válido (mínimo 8 dígitos).";
   326→      }
   327→      if (form.notes.length > 500) encontrados.notes = "Las notas no pueden superar los 500 caracteres.";
   328→      if (!form.consent) encontrados.consent = "Necesitamos tu autorización para coordinar la cita.";
   329→    }
   330→    return encontrados;
   331→  };
   332→
   333→  const avanzar = () => {
   334→    const encontrados = validarPaso(paso);
   335→    if (Object.keys(encontrados).length > 0) {
   336→      setErrores(encontrados);
   337→      return;
   338→    }
   339→    setErrores({});
   340→    setPaso((prev) => Math.min(PASOS.length, prev + 1));
   341→  };
   342→
   343→  const retroceder = () => {
   344→    setErrores({});
   345→    setPaso((prev) => Math.max(1, prev - 1));
   346→  };
   347→
   348→  /* Enter en pasos 1–3 solo avanza (nunca envía); en el paso 4 envía. */
   349→  const enviarReserva = async (evento: FormEvent<HTMLFormElement>) => {
   350→    evento.preventDefault();
   351→    if (paso < PASOS.length) {
   352→      avanzar();
   353→      return;
   354→    }
   355→
   356→    const encontrados = validarPaso(4);
   357→    if (Object.keys(encontrados).length > 0) {
   358→      setErrores(encontrados);
   359→      return;
   360→    }
   361→    setErrores({});
   362→    setEnviando(true);
   363→
   364→    try {
   365→      const payload: AppointmentInput = {
   366→        petName: form.petName.trim(),
   367→        species: form.species as Especie,
   368→        breed: form.breed.trim(),
   369→        ageYears: form.ageYears,
   370→        service: form.service,
   371→        preferredVet: form.preferredVet,
   372→        date: form.date,
   373→        timeSlot: form.timeSlot,
   374→        tutorName: form.tutorName.trim(),
   375→        email: form.email.trim(),
   376→        phone: form.phone.trim(),
   377→        notes: form.notes.trim(),
   378→        consent: true,
   379→      };
   380→
   381→      const res = await fetch("/api/appointments", {
   382→        method: "POST",
   383→        headers: { "Content-Type": "application/json" },
   384→        body: JSON.stringify(payload),
   385→      });
   386→      const json = (await res.json().catch(() => null)) as {
   387→        ok?: boolean;
   388→        code?: string;
   389→        error?: string;
   390→        fieldErrors?: Record<string, string>;
   391→        appointment?: { date: string; timeSlot: string; service: string; petName: string; tutorName: string };
   392→      } | null;
   393→
   394→      if (res.status === 409) {
   395→        // La franja se ocupó mientras agendábamos: volver al paso 3 y refrescar.
   396→        toast.error(json?.error ?? "Lo sentimos, esa franja acaba de ser reservada. Elige otro horario.");
   397→        setForm((prev) => ({ ...prev, timeSlot: "" }));
   398→        setErrores((prev) => {
   399→          const siguientes = { ...prev };
   400→          delete siguientes.timeSlot;
   401→          return siguientes;
   402→        });
   403→        setPaso(3);
   404→        if (form.date) void consultarDisponibilidad(form.date);
   405→        return;
   406→      }
   407→
   408→      if (!res.ok || !json?.ok || !json.code || !json.appointment) {
   409→        if (json?.fieldErrors) setErrores(json.fieldErrors);
   410→        toast.error(json?.error ?? "No pudimos registrar tu reserva. Revisa los datos e intenta de nuevo.");
   411→        return;
   412→      }
   413→
   414→      setConfirmacion({ code: json.code, ...json.appointment });
   415→      toast.success("¡Reserva confirmada! Te esperamos.");
   416→    } catch {
   417→      toast.error("Tuvimos un problema de conexión. Verifica tu internet e intenta de nuevo.");
   418→    } finally {
   419→      setEnviando(false);
   420→    }
   421→  };
   422→
   423→  const reiniciar = () => {
   424→    setForm(FORM_INICIAL);
   425→    setErrores({});
   426→    setPaso(1);
   427→    setDisponibilidad(null);
   428→    setErrorDispo(false);
   429→    setCargandoDispo(false);
   430→    setConfirmacion(null);
   431→  };
   432→
   433→  const servicioElegido = services.find((servicio) => servicio.id === form.service);
   434→  const profesionalElegido = vets.find((vet) => vet.id === form.preferredVet);
   435→  const tituloPasoActual = PASOS[paso - 1]?.titulo ?? "";
   436→  const pasoActual = confirmacion ? PASOS.length : paso;
   437→
   438→  /* Enlace de WhatsApp con confirmación prellenada (solo tras confirmar). */
   439→  const waConfirmHref = confirmacion
   440→    ? `https://wa.me/${site.whatsappNumber}?text=${encodeURIComponent(
   441→        `Hola LONGIVET, confirmo mi cita ${confirmacion.code} para ${confirmacion.petName} el ${formatFechaLarga(
   442→          confirmacion.date,
   443→        )} a las ${confirmacion.timeSlot}.`,
   444→      )}`
   445→    : "#";
   446→
   447→  return (
   448→    <MotionConfig reducedMotion="user">
   449→      <section id="agendar" aria-labelledby="titulo-agendar" className="bg-background py-20 lg:py-24">
   450→        <div className="mx-auto max-w-7xl px-4 sm:px-6 lg:px-8">

   453→            <p className="text-sm font-bold uppercase tracking-[0.2em] text-brand-teal">

   460→              Agenda la cita de tu compañero
